def haversine_distance(lat1: float, lon1: float, lat2: float, lon2: float) -> float:
    R = 6371000
    phi1, phi2 = math.radians(lat1), math.radians(lat2)
    delta_phi = math.radians(lat2 - lat1)
    delta_lambda = math.radians(lon2 - lon1)
    a = math.sin(delta_phi / 2) ** 2 + math.cos(phi1) * math.cos(phi2) * math.sin(delta_lambda / 2) ** 2
    return R * (2 * math.atan2(math.sqrt(a), math.sqrt(1 - a)))

def resolve_municipal_region(location_str: str):
    text = location_str.lower()
    if "anna salai" in text or "teynampet" in text:
        return "Chennai Central", "Zone 9 - Teynampet", "Ward 118"
    elif "t nagar" in text or "kodambakkam" in text:
        return "Chennai Central", "Zone 10 - Kodambakkam", "Ward 134"
    elif "adyar" in text or "besant nagar" in text:
        return "Chennai South", "Zone 13 - Adyar", "Ward 174"
    return "Chennai Central", "Zone 8 - Anna Nagar", "Ward 102"

@app.patch("/api/reports/{report_id}/status")
async def update_report_status(report_id: str, payload: StatusUpdateRequest):
    for r in REPORTS_DB:
        if r["id"] == report_id:
            r["status"] = payload.status
            return {"status": "success", "updated_id": report_id, "new_status": r["status"]}
    return {"status": "error", "message": "Report not found"}

@app.post("/api/report")
async def create_report(
    issue_type: str = Form(...),
    description: str = Form(...),
    lat: float = Form(13.0827),
    lng: float = Form(80.2707),
    location_name: str = Form("Chennai, Tamil Nadu"),
    file: UploadFile = File(...),
):
    # Read file content safely
    _ = await file.read()

    region, zone, ward = resolve_municipal_region(location_name)

    # Check for duplicates within 35 meters
    duplicate_found = any(
        haversine_distance(lat, lng, r["lat"], r["lng"]) <= 35.0 and r["issue_type"].lower() == issue_type.lower()
        for r in REPORTS_DB
    )

    desc_lower = description.lower()
    is_high = any(w in desc_lower for w in ["deep", "huge", "large", "danger", "accident", "broken"])
    severity = "High" if is_high else "Medium"
    priority = 88 if is_high else 64

    new_report = {
        "id": f"CIVIA-2026-{uuid.uuid4().hex[:4].upper()}",
        "issue_type": issue_type,
        "description": description,
        "region": region,
        "zone": zone,
        "ward": ward,
        "lat": lat,
        "lng": lng,
        "location_name": location_name,
        "severity": severity,
        "confidence": 0.94,
        "priority_score": priority,
        "ai_notes": "Automated vision triage classified road surface damage and assigned immediate priority.",
        "duplicate_detected": duplicate_found,
        "status": "IN PROGRESS",
        "created_at": datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
    }

    # Prepend new reports so they appear at the top
    REPORTS_DB.insert(0, new_report)

    return {"status": "success", "report": new_report}
